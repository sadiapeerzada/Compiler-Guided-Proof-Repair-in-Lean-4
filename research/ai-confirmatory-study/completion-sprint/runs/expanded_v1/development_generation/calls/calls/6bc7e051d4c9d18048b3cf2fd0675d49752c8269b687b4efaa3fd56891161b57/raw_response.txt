STATEMENT: For any integers \(a\), \(b\), and \(c\) where \(b > 0\) and \(c > 0\), the remainder of \(a\) divided by \(b\) is less than the remainder of \(a\) divided by \(c\).

LEAN:
```lean
theorem sprint_development_0017 (a b c : ℤ) (hb : b > 0) (hc : c > 0) : a % b < a % c := by
  unfold emod
  have h1 : a % b = a - b * (a / b) := by norm_num
  have h2 : a % c = a - c * (a / c) := by norm_num
  have h3 : a / b ≤ a / c := by linarith [hb, hc]
  have h4 : b * (a / b) ≤ c * (a / c) := mul_le_mul_of_le_of_le h3 hb hc
  have h5 : a - b * (a / b) < a - c * (a / c) := sub_lt_sub_of_le_of_lt h4 h1 h2
  simp [h5]
```
