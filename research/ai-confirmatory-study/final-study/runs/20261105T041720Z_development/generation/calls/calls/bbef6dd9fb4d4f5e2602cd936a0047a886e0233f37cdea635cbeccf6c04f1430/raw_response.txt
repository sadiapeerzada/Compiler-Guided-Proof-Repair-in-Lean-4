STATEMENT: For all natural numbers \(a\) and \(b\) where \(b > 0\), the remainder \(r\) when \(a\) is divided by \(b\) satisfies \(0 \leq r < b\).

LEAN:
```lean
theorem final_dev_0118 (a b : Nat) (h : b > 0) : a % b < b ∧ 0 ≤ a % b := by
  classical
  have : ∃ r, a = b * (a / b) + r := by
    exact Nat.div_and_mod a b
  cases this with r eq
  have : 0 ≤ r := by
    have : a / b * b ≤ a := by
      rw [eq] at h ; exact Nat.mul_le_mul_left h zero_le_one
    rw [Nat.div_le_div_of_pos h] at this
    exact this
  have : r < b := by
    have : a / b * b + r = a := by
      rw [eq]
    have : a / b * b ≤ a := by
      rw [this] ; exact zero_le_one
    have : a / b * b + r ≤ a + b - 1 := by
      rw [Nat.add_assoc, Nat.add_comm (a / b * b), Nat.add_left_comm b, Nat.add_left_comm (a / b * b)]
      exact Nat.le_add_right
    have : r ≤ b - 1 := by
      rw [Nat.add_sub_cancel_right (a / b * b)] at this
      exact Nat.sub_le_sub_left this h
    exact Nat.lt_of_le_of_lt this zero_lt_one
  exact ⟨r, this⟩
```
