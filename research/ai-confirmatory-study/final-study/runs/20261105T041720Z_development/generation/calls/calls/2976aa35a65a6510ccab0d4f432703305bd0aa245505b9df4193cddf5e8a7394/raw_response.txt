STATEMENT: For any finite sequence of real numbers \(a_1, a_2, \ldots, a_n\), the sum of the sequence can be expressed as the sum of the first \(k\) terms plus the sum of the remaining terms, where \(1 \leq k < n\).

LEAN:
```lean
theorem final_dev_0007 (a : ℕ → ℝ) (n : ℕ) (k : ℕ) (hkn : k < n) : 
  ∑ i in finset.range n, a i = ∑ i in finset.range k, a i + ∑ i in finset.range (n - k), a (i + k) := by
  have h1 : ∑ i in finset.range n, a i = ∑ i in finset.range k, a i + ∑ i in finset.range (n - k), a (i + k) := by
    rw [finset.sum_range_add, finset.sum_range_add]
    rw [finset.sum_range, finset.sum_range, finset.sum_shift]
    apply finset.sum_congr
    intro i
    have h2 : i < n ↔ i < k ∨ i ≥ k ∧ i < n := by
      split
      intro h
      cases le_total i k with h1 h2
      · exact h1 ▸ h
      · exact h2 ▸ h
      intro h
      cases h with h1 h2
      · exact h1 ▸ h
      · exact h2 ▸ h
    rw [h2]
    simp
  exact h1
```
