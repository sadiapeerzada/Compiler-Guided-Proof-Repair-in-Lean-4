STATEMENT: For all integers \(a\) and \(b\), if \(a^2\) is divisible by 4 and \(b^2\) is divisible by 4, then \((a + b)^2\) is divisible by 4.

  have h2' : ∃ k : Int, b^2 = 4 * k := by

  obtain ⟨k, hk⟩ := h1'
  obtain ⟨l, hl⟩ := h2'
  have : (a + b)^2 = 4 * (k + l + 2 * a * b / 4) := by
    ring
  exact Int.mul_right_inj.1 this
