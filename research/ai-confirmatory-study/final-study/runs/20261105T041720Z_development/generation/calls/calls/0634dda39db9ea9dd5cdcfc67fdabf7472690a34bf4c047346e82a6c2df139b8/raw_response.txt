STATEMENT: For any list `l` of natural numbers, the length of the list after filtering out elements that are greater than the first element of the list is at most the length of the original list.

LEAN:
```lean
theorem final_dev_0058 (l : List Nat) : List.length (List.filter (fun x => x ≤ l.head) l) ≤ List.length l := by
  induction l with
  | nil => simp
  | cons x xs ih =>
    have h1 : x ≤ x := le_refl x
    have h2 : List.filter (fun y => y ≤ x) (List.nil) = List.nil := by simp
    have h3 : List.filter (fun y => y ≤ x) (xs ++ List.nil) = List.filter (fun y => y ≤ x) xs := by simp
    have h4 : List.filter (fun y => y ≤ x) xs ≤ List.length xs := by
      simp [List.filter_le_length]
    have h5 : List.length (xs ++ List.nil) = List.length xs := by simp
    have h6 : List.filter (fun y => y ≤ x) (l ++ List.nil) = List.filter (fun y => y ≤ x) l := by simp
    have h7 : List.length (List.filter (fun y => y ≤ x) l) ≤ List.length l := by
      simp [ih, h5, h6]
    exact h7
```
