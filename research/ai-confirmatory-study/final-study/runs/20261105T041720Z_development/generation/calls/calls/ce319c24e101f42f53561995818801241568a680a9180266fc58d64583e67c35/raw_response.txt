STATEMENT: For any integer \( n \), if \( n^2 \) is divisible by 4, then \( n \) is even.

theorem final_dev_0159 (n : ℤ) (h : n^2 % 4 = 0) : even n := by
  obtain a : ℤ := n / 2
  have h1 : n = 2 * a := by rw [← mul_two, Int.cast_eq_zero.mp h]
  have h2 : n^2 = 4 * a^2 := by rw [h1, mul_assoc, two_mul, mul_assoc, h1, h1, mul_comm, mul_assoc, two_mul, h1]
  have h3 : 4 * a^2 % 4 = 0 := by simp
  have h4 : a^2 % 1 = 0 := by rw [h3, Int.cast_eq_zero]
  have h5 : a^2 = 0 ∨ a^2 = 1 ∨ a^2 = 2 ∨ a^2 = 3 := by aesop
  have h6 : a^2 = 0 ∨ a^2 = 1 ∨ a^2 = 2 ∨ a^2 = 3 → a^2 % 1 = 0 := by
    intro ha
    · exact Int.cast_eq_zero.1 (by linarith)
    · exact Int.cast_eq_one.1 (by linarith)
    · exact Int.cast_eq_two.1 (by linarith)
    · exact Int.cast_eq_three.1 (by linarith)
  have h7 : a^2 % 1 = 0 := by exact h5 ▸ h6
  have h8 : a % 1 = 0 ∨ a % 1 = 1 ∨ a % 1 = 2 ∨ a % 1 = 3 := by aesop
  have h9 : a % 1 = 0 ∨ a % 1 = 1 ∨ a % 1 = 2 ∨ a % 1 = 3 → a % 2 = 0 := by
    intro ha
    · exact Int.cast_eq_zero.1 (by linarith)
    · exact Int.cast_eq_one.1 (by linarith)
    · exact Int.cast_eq_two.1 (by linarith)
    · exact Int.cast_eq_three.1 (by linarith)
  have h10 : a % 2 = 0 := by exact h8 ▸ h9
  exact h10 ▸ even_of_int_cast
