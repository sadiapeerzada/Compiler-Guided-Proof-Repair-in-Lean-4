STATEMENT: For any integers \(a\) and \(b\), the absolute value of their difference is less than or equal to the sum of their absolute values.

LEAN:
```lean
theorem final_dev_0004 (a b : Int) : |a - b| ≤ |a| + |b| := by
  have h1 : |a - b| = |b - a| := by simp [Int.abs_sub_symm]
  have h2 : |a - b| ≤ |a| + |b - a| := by
    by_cases h: a - b ≥ 0
    · have : a - b = |a - b| := by simp [h]
      have : |a - b| ≤ |a| + (b - a) := by
        have : b - a = - (a - b) := by simp
        rw [this, Int.abs_neg, Int.abs_add]
        ring
    · have : a - b = - (b - a) := by simp [h]
      have : |a - b| ≤ |a| + (b - a) := by
        have : b - a = - (a - b) := by simp
        rw [this, Int.abs_neg, Int.abs_add]
        ring
  exact h2
```
