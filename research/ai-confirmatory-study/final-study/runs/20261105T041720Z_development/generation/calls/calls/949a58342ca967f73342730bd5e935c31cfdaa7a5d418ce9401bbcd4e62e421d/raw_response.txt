STATEMENT: For any integer \( n \), if \( n^2 \) is divisible by 4, then \( n \) is even.

  obtain ⟨k, hk⟩ := int.exists_eq_add_one_of_even n
  have h1 : (2 * k + 1)^2 % 4 ≠ 0 := by

  have h2 : n^2 % 4 = 0 := by
    exact h
  linarith [hk, h1, h2]
