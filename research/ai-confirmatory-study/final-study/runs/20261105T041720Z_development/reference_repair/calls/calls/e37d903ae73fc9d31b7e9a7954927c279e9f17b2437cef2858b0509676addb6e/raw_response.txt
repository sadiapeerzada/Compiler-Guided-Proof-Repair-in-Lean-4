sorry final_dev_0132 {A B : Type*} [fintype A] [fintype B] {f : A → B} (X Y : set A) : f '' (X ∪ Y) ⊆ f '' X ∪ f '' Y :=
begin
  intro b,
  obtain h : ∃ a, a ∈ X ∪ Y ∧ f a = b := by
    use some a,
    cases a with a ha,
    exact ⟨a, ha, by simp⟩,
  obtain (hX : a ∈ X) | (hY : a ∈ Y) := h.2,
  (obtain hX' : f '' X ∩ {b} ≠ ∅ := by
      exact ⟨⟨_, hX⟩, by simp⟩),
  (obtain hX'' : f '' X ∩ {b} = {b} := by
      exact hX'.1),
  (obtain hY' : f '' Y ∩ {b} ≠ ∅ := by
      exact ⟨⟨_, hY⟩, by simp⟩),
  (obtain hY'' : f '' Y ∩ {b} = {b} := by
      exact hY'.1),
  exact (hX''.1) (hY''.1)
end.
