by
  induction l with
  | nil => simp [List.sum_nil, List.tail_nil, List.headOption_nil, List.get_or_else_zero]
  | cons h t ih =>
    simp [List.sum_cons, List.tail_cons, List.headOption_cons, List.get_or_else_cons, ih]
    (h + (t.sum + List.get_or_else_zero)) = (h + (t.tail.sum + List.get_or_else_zero)) + List.get_or_else_zero
  exact (h + (t.tail.sum + List.get_or_else_zero)) + List.get_or_else_zero = (h + (t.tail.sum + List.get_or_else_zero)) + List.get_or_else_zero

END BY
